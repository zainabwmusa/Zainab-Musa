
  <script>
    alert('Need a website ? DM +2347011766328');

    // Responsive menu toggle
    const menuToggle = document.getElementById('menu-toggle');
    const navLinks = document.querySelector('.nav');
    const menuSideText = document.getElementById('menu-side-text');
    // Hide nav by default on mobile
    if (window.innerWidth <= 1024) {
      navLinks.classList.remove('show');
      menuSideText.style.display = 'none';
    }
    menuToggle.addEventListener('click', function() {
      navLinks.classList.toggle('show');
      // Show side text only if menu is open and on small screens
      if (window.innerWidth <= 1024 && navLinks.classList.contains('show')) {
        menuSideText.style.display = 'block';
      } else {
        menuSideText.style.display = 'none';
      }
    });
    // Hide side text and nav if window is resized up
    window.addEventListener('resize', function() {
      if (window.innerWidth > 1024) {
        navLinks.classList.remove('show');
        menuSideText.style.display = 'none';
      } else {
        navLinks.classList.remove('show');
        menuSideText.style.display = 'none';
      }
    });
    // Hide nav and side text when a nav link is clicked (mobile UX)
    document.querySelectorAll('.nav ul a').forEach(link => {
      link.addEventListener('click', function() {
        if (window.innerWidth <= 1024) {
          navLinks.classList.remove('show');
          menuSideText.style.display = 'none';
        }
      });
    });
  </script>
